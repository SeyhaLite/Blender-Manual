  អំពីកម្មវិធីសេរី និងអាជ្ញាបណ្ណ GPL
###################################################

.. figure:: /images/getting-started_about_license_gnu-logo.png
   :align: right

នៅពេលដែលគេលឺពាក្យ «កម្មវិធីសេរី» (free software) អ្វីដែលចូលមកក្នុងគំនិតមុនគេ
ប្រហែលជា «គ្មានថ្លៃ»។ ទោះបីជាការនេះជារឿយៗជាការពិតក៏ដោយ ពាក្យ «កម្មវិធីសេរី»
ដូចដែល Free Software Foundation (អ្នកបង្កើតគម្រោង GNU និងអ្នកនិពន្ធ GNU General Public License)
ប្រើ មានន័យថា «សេរីដូចជាសេរីភាព» (free as in freedom) មិនមែនក្នុងន័យ «គ្មានថ្លៃ»
(ដែលជាធម្មតាហៅថា «free as in free beer» ឬ *gratis*) ទេ។
កម្មវិធីសេរីក្នុងន័យនេះ គឺជាកម្មវិធីដែលអ្នកមានសេរីភាពក្នុងការប្រើ ចម្លង កែប្រែ
និងចែកចាយបន្ត ដោយគ្មានកំណត់។ ចូរប្រៀបធៀបជាមួយអាជ្ញាបណ្ណនៃកញ្ចប់កម្មវិធីពាណិជ្ជកម្មភាគច្រើន
ដែលអ្នកត្រូវបានអនុញ្ញាតឱ្យដំឡើងកម្មវិធីលើកុំព្យូទ័រតែមួយគ្រឿង មិនអនុញ្ញាតឱ្យចម្លងទេ
ហើយមិនដែលបានឃើញកូដប្រភពឡើយ។ កម្មវិធីសេរីផ្ដល់សេរីភាពដ៏អស្ចារ្យដល់អ្នកប្រើប្រាស់ចុងក្រោយ។
ដោយសារកូដប្រភពមានជាសកល ក៏មានឱកាសច្រើនជាងមុនដែរ ក្នុងការរកឃើញ និងកែកំហុស (bug)។

នៅពេលដែលកម្មវិធីមួយត្រូវបានផ្ដល់អាជ្ញាបណ្ណក្រោម GNU General Public License (GPL)៖

- អ្នកមានសិទ្ធិប្រើកម្មវិធីនោះក្នុងគោលបំណងណាមួយក៏បាន។
- អ្នកមានសិទ្ធិកែប្រែកម្មវិធី និងទទួលបានកូដប្រភព។
- អ្នកមានសិទ្ធិចម្លង និងចែកចាយកម្មវិធី។
- អ្នកមានសិទ្ធិធ្វើឱ្យកម្មវិធីប្រសើរឡើង និងចេញផ្សាយកំណែផ្ទាល់ខ្លួនរបស់អ្នក។

ជាការតបស្នងចំពោះសិទ្ធិទាំងនេះ អ្នកមានទំនួលខុសត្រូវមួយចំនួន ប្រសិនបើអ្នកចែកចាយកម្មវិធីមួយ
ដែលមានអាជ្ញាបណ្ណ GPL។ ទំនួលខុសត្រូវទាំងនេះត្រូវបានរចនាឡើងដើម្បីការពារសេរីភាពរបស់អ្នក
និងសេរីភាពរបស់អ្នកដទៃ៖

- អ្នកត្រូវតែផ្ដល់ច្បាប់ចម្លងនៃ GPL ជាមួយកម្មវិធី ដើម្បីឱ្យអ្នកទទួលដឹងពីសិទ្ធិរបស់ខ្លួន
  ក្រោមអាជ្ញាបណ្ណនោះ។
- អ្នកត្រូវតែរួមបញ្ចូលកូដប្រភព ឬធ្វើឱ្យកូដប្រភពនោះអាចទទួលបានដោយសេរី។
- ប្រសិនបើអ្នកកែប្រែកូដ ហើយចែកចាយកំណែដែលបានកែប្រែ អ្នកត្រូវតែផ្ដល់អាជ្ញាបណ្ណ
  GPL (ឬអាជ្ញាបណ្ណដែលឆបគ្នា) ដល់ការកែប្រែរបស់អ្នក។
- អ្នកមិនអាចដាក់កំហិតលើអាជ្ញាបណ្ណរបស់កម្មវិធីលើសពីលក្ខខណ្ឌរបស់ GPL
  (អ្នកមិនអាចបំលែងកម្មវិធី GPL ទៅជាផលិតផលកម្មសិទ្ធិឯកជនបានទេ)។

សម្រាប់ព័ត៌មានបន្ថែមអំពី GPL សូមមើលទំព័ររបស់វានៅលើ
`គេហទំព័រគម្រោង GNU <https://www.gnu.org/licenses/licenses.html#GPL>`__។
